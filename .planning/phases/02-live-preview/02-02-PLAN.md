**Create `src/highlighter.rs` with a `MarkdownHighlighter` struct** that provides line-by-line highlighting info:

```rust
use syntect::parsing::SyntaxSet;
use syntect::highlighting::{ThemeSet, Theme, HighlightLines};
use syntect_tui::into_span;
use ratatui::text::{Line, Span};

pub struct MarkdownHighlighter {
    syntax_set: SyntaxSet,
    theme: Theme,
}

impl MarkdownHighlighter {
    pub fn new() -> Self {
        let ss = SyntaxSet::load_defaults_newlines();
        let ts = ThemeSet::load_defaults();
        // Use base16-ocean.dark — terminal-compatible theme (D-08)
        let theme = ts.themes["base16-ocean.dark"].clone();
        MarkdownHighlighter {
            syntax_set: ss,
            theme,
        }
    }

    /// Highlight markdown text, returning styled Lines.
    pub fn highlight_lines(&self, text: &str) -> Vec<Line<'static>> {
        let syntax = self.syntax_set
            .find_syntax_by_extension("md")
            .unwrap_or_else(|| self.syntax_set.find_syntax_plain_text());

        let mut h = HighlightLines::new(syntax, &self.theme);

        text.lines()
            .map(|line| {
                let ranges = h.highlight_line(line, &self.syntax_set)
                    .unwrap_or_default();
                let spans: Vec<Span<'static>> = ranges
                    .into_iter()
                    .map(|(style, text)| into_span((style, text)))
                    .collect();
                Line::from(spans)
            })
            .collect()
    }
}
```
