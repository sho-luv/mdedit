**1. Create `App::execute_vim_command()` method** that translates VimCommand variants into textarea operations:

```rust
fn execute_vim_command(&mut self, cmd: VimCommand) {
    match cmd {
        VimCommand::Move(cursor_move) => {
            self.editor.textarea_mut().move_cursor(cursor_move);
        }
        VimCommand::MoveN(cursor_move, n) => {
            for _ in 0..n {
                self.editor.textarea_mut().move_cursor(cursor_move);
            }
        }
        VimCommand::Delete { motion } => {
            self.execute_vim_operator(Operator::Delete, motion);
        }
        VimCommand::Change { motion } => {
            self.execute_vim_operator(Operator::Change, motion);
        }
        VimCommand::Yank { motion } => {
            self.execute_vim_operator(Operator::Yank, motion);
        }
        VimCommand::DeleteChar => {
            let count = 1; // count already applied by handler
            for _ in 0..count {
                self.editor.textarea_mut().delete_next_char();
            }
            self.mark_content_changed();
        }
        VimCommand::PasteAfter => {
            if let Some(ref handler) = self.vim_handler {
                let text = handler.yank_register.clone();
                if !text.is_empty() {
                    // Move cursor right (paste after), then insert
                    self.editor.textarea_mut().move_cursor(CursorMove::Forward);
                    self.editor.textarea_mut().insert_str(&text);
                    self.mark_content_changed();
                }
            }
        }
        VimCommand::PasteBefore => {
            if let Some(ref handler) = self.vim_handler {
                let text = handler.yank_register.clone();
                if !text.is_empty() {
                    self.editor.textarea_mut().insert_str(&text);
                    self.mark_content_changed();
                }
            }
        }
        VimCommand::EnterInsert(pos) => {
            // Position cursor based on InsertPosition before entering insert mode
            match pos {
                InsertPosition::BeforeCursor => {} // cursor stays
                InsertPosition::AfterCursor => {
                    self.editor.textarea_mut().move_cursor(CursorMove::Forward);
                }
                InsertPosition::LineStart => {
                    self.editor.textarea_mut().move_cursor(CursorMove::Head);
                }
                InsertPosition::LineEnd => {
                    self.editor.textarea_mut().move_cursor(CursorMove::End);
                }
                InsertPosition::NewLineBelow => {
                    self.editor.textarea_mut().move_cursor(CursorMove::End);
                    self.editor.textarea_mut().insert_newline();
                    self.mark_content_changed();
                }
                InsertPosition::NewLineAbove => {
                    self.editor.textarea_mut().move_cursor(CursorMove::Head);
                    self.editor.textarea_mut().insert_newline();
                    self.editor.textarea_mut().move_cursor(CursorMove::Up);
                    self.mark_content_changed();
                }
            }
            self.mode = AppMode::Insert;
        }
        VimCommand::ExitInsert => {
            self.mode = AppMode::Normal;
        }
        VimCommand::Undo => {
            self.editor.textarea_mut().undo();
            self.mark_content_changed();
        }
        VimCommand::Redo => {
            self.editor.textarea_mut().redo();
            self.mark_content_changed();
        }
        VimCommand::EnterSearch => {
            // Reuse existing search infrastructure (D-25)
            self.search_query.clear();
            self.search_cursor_before = self.editor.cursor_position();
            self.search_match_index = 0;
            self.search_match_count = 0;
            self.mode = AppMode::Search;
        }
        VimCommand::EnterCommand => {
            self.mode = AppMode::Command;
        }
        VimCommand::ExitCommand => {
            self.mode = AppMode::Normal;
        }
        VimCommand::CommandExecute(ref cmd_str) => {
            self.execute_ex_command(cmd_str);
            // Mode already set to Normal by handler
            self.mode = AppMode::Normal;
        }
        VimCommand::Save => {
            if self.editor.filepath().is_some() {
                self.do_save();
            } else {
                self.filename_input = String::new();
                self.quit_after_save = false;
                self.mode = AppMode::PromptFilename;
            }
        }
        VimCommand::Quit { force } => {
            if force || !self.editor.is_modified() {
                self.should_quit = true;
            } else {
                self.mode = AppMode::ConfirmQuit;
            }
        }
        VimCommand::SaveAndQuit => {
            if self.editor.filepath().is_some() {
                if self.do_save() {
                    self.should_quit = true;
                }
            } else {
                self.filename_input = String::new();
                self.quit_after_save = true;
                self.mode = AppMode::PromptFilename;
            }
        }
        VimCommand::EnterVisual { line_wise } => {
            self.editor.textarea_mut().start_selection();
            self.mode = AppMode::Visual;
        }
        VimCommand::ExitVisual => {
            self.editor.textarea_mut().cancel_selection();
            self.mode = AppMode::Normal;
        }
        VimCommand::None | VimCommand::CommandAppend(_) | VimCommand::CommandBackspace => {}
        // Visual operators handled in Plan 03
        _ => {}
    }
}
```

**2. Create `App::execute_vim_operator()` helper** for d/c/y + motion:

fn execute_vim_operator(&mut self, op: Operator, motion: Motion) {
    match motion {
        Motion::Line => {
            // dd/yy/cc: operate on full line(s)
            self.editor.delete_or_yank_lines(op, count);
        }
        Motion::ToEnd => {
            // D/C/d$/c$: select to end of line, operate
            self.editor.textarea_mut().start_selection();
            self.editor.textarea_mut().move_cursor(CursorMove::End);
            self.operate_on_selection(op);
        }
        Motion::ToStart => {
            // d0: select to start of line, operate
            self.editor.textarea_mut().start_selection();
            self.editor.textarea_mut().move_cursor(CursorMove::Head);
            self.operate_on_selection(op);
        }
        // For other motions: start selection, move, then operate
        other => {
            let cursor_move = motion_to_cursor_move(&other);
            self.editor.textarea_mut().start_selection();
            self.editor.textarea_mut().move_cursor(cursor_move);
            self.operate_on_selection(op);
        }
    }
}

fn handle_vim_key(&mut self, key: KeyEvent) {
    // Ctrl+P still toggles layout (intercept before vim)
    if key.modifiers == KeyModifiers::CONTROL && key.code == KeyCode::Char('p') {
        self.layout_mode = self.layout_mode.next();
        self.status_bar.set_message(self.layout_mode.label());
        return;
    }
    if let Some(ref mut handler) = self.vim_handler {
        let cmd = handler.handle_key(key);
        // Need to drop the borrow before execute
        drop(handler);  // This won't work with Option ref. Instead:
    }
    // Pattern: extract command, then execute
    let cmd = self.vim_handler.as_mut().map(|h| h.handle_key(key)).unwrap_or(VimCommand::None);
    self.execute_vim_command(cmd);
}

fn execute_ex_command(&mut self, cmd: &str) {
    match cmd.trim() {
        "w" => { self.do_save(); }
        "q" => {
            if self.editor.is_modified() {
                self.status_bar.set_message("No write since last change (add ! to override)");
            } else {
                self.should_quit = true;
            }
        }
        "q!" => { self.should_quit = true; }
        "wq" | "x" => {
            if self.do_save() { self.should_quit = true; }
        }
        other => {
            self.status_bar.set_message(&format!("Not a command: {}", other));
        }
    }
}
